# Simple TODO App Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a self-contained browser TODO app in a single `index.html` file with localStorage persistence.

**Architecture:** One `index.html` with inline `<style>` and `<script>`. A single in-memory `todos` array is the source of truth; every change calls `save()` then `render()`. No build step, no dependencies, no server.

**Tech Stack:** Plain HTML, CSS, and vanilla JavaScript (ES6). `localStorage` for persistence.

## Global Constraints

- Single file only: everything lives in `index.html` (inline CSS + JS). No external files, no libraries, no build step.
- Persistence via `localStorage` under one key, storing a JSON array of `{ id, text, completed, dueDate }`.
- `dueDate` is an ISO `YYYY-MM-DD` string or `null`.
- State is the single source of truth: UI is always produced by `render()` from the `todos` array.
- Look & feel: clean, minimal, neutral light theme, centered card, system font, one quiet accent color, mobile-usable.
- "Today" for overdue comparison uses the local date.

**Testing note:** There is no test framework (no build/deps per spec). Each task's verification is manual: open `index.html` in a browser (or refresh) and confirm the described behavior, including persistence across a refresh.

### Task 2: Render the list and active count

**Files:**
- Modify: `index.html` (add to `<script>`)

**Interfaces:**
- Consumes: `todos`, `filter`, DOM ids `#list`, `#count` from Task 1.
- Produces:
  - `function isOverdue(t)` — returns `true` if `t.dueDate` is set, `t` is not completed, and `t.dueDate < todayStr()`.
  - `function todayStr()` — returns today's local date as `YYYY-MM-DD`.
  - `function visibleTodos()` — returns `todos` filtered by `filter`.
  - `function render()` — rebuilds `#list` from `visibleTodos()` and updates `#count` with the number of active (incomplete) tasks; shows an `.empty` message when the filtered list is empty. Each `<li>` has `data-id`, a checkbox, a `.text` span, a `.due` span (when `dueDate` set), and a `.del` button.

- [ ] **Step 1: Add render logic to the script (before `load();`)**

```javascript
  function todayStr() {
    const d = new Date();
    const m = String(d.getMonth() + 1).padStart(2, "0");
    const day = String(d.getDate()).padStart(2, "0");
    return `${d.getFullYear()}-${m}-${day}`;
  }
  function isOverdue(t) {
    return !!t.dueDate && !t.completed && t.dueDate < todayStr();
  }
  function visibleTodos() {
    if (filter === "active") return todos.filter(t => !t.completed);
    if (filter === "completed") return todos.filter(t => t.completed);
    return todos;
  }
  function render() {
    const list = document.getElementById("list");
    list.innerHTML = "";
    const items = visibleTodos();
    if (items.length === 0) {
      const li = document.createElement("li");
      li.className = "empty";
      li.textContent = "Nothing here.";
      list.appendChild(li);
    }
    for (const t of items) {
      const li = document.createElement("li");
      li.dataset.id = t.id;
      if (t.completed) li.classList.add("completed");
      if (isOverdue(t)) li.classList.add("overdue");

      const cb = document.createElement("input");
      cb.type = "checkbox";
      cb.checked = t.completed;
      cb.className = "toggle";

      const text = document.createElement("span");
      text.className = "text";
      text.textContent = t.text;

      li.appendChild(cb);
      li.appendChild(text);

      if (t.dueDate) {
        const due = document.createElement("span");
        due.className = "due";
        due.textContent = "Due " + t.dueDate;
        li.appendChild(due);
      }

      const del = document.createElement("button");
      del.className = "del";
      del.textContent = "×";
      del.title = "Delete";
      li.appendChild(del);

      list.appendChild(li);
    }
    const active = todos.filter(t => !t.completed).length;
    document.getElementById("count").textContent =
      `${active} item${active === 1 ? "" : "s"} left`;
  }
```

- [ ] **Step 1: Add `addTask` and wire up events (before `load();`)**

```javascript
  function addTask() {
    const input = document.getElementById("new-task");
    const dueInput = document.getElementById("new-due");
    const text = input.value.trim();
    if (!text) return;
    todos.push({ id: uid(), text, completed: false, dueDate: dueInput.value || null });
    input.value = "";
    dueInput.value = "";
    input.focus();
    save();
    render();
  }

  document.getElementById("add-btn").addEventListener("click", addTask);
  document.getElementById("new-task").addEventListener("keydown", (e) => {
    if (e.key === "Enter") addTask();
  });
```

### Task 4: Toggle complete and delete

**Files:**

**Interfaces:**
- Consumes: `todos`, `save`, `render`, `#list`.
- Produces: a delegated `click`/`change` listener on `#list` that toggles `completed` when a `.toggle` checkbox changes and removes the task when a `.del` button is clicked, finding the task by the `li`'s `data-id`.

- [ ] **Step 1: Add delegated handlers (before `load();`)**

```javascript
  const list = document.getElementById("list");
  list.addEventListener("change", (e) => {
    if (!e.target.classList.contains("toggle")) return;
    const id = e.target.closest("li").dataset.id;
    const t = todos.find(t => t.id === id);
    if (t) { t.completed = e.target.checked; save(); render(); }
  });
  list.addEventListener("click", (e) => {
    if (!e.target.classList.contains("del")) return;
    const id = e.target.closest("li").dataset.id;
    todos = todos.filter(t => t.id !== id);
    save();
    render();
  });
```

```javascript
  function startEdit(li, t) {
    li.innerHTML = "";
    const edit = document.createElement("input");
    edit.type = "text";
    edit.className = "edit";
    edit.value = t.text;

    const dueEdit = document.createElement("input");
    dueEdit.type = "date";
    dueEdit.className = "due-edit";
    dueEdit.value = t.dueDate || "";

    let done = false;
    function commit() {
      if (done) return;
      done = true;
      const newText = edit.value.trim();
      if (!newText) {
        todos = todos.filter(x => x.id !== t.id);
      } else {
        t.text = newText;
        t.dueDate = dueEdit.value || null;
      }
      save();
      render();
    }
    function cancel() { if (done) return; done = true; render(); }

    edit.addEventListener("keydown", (e) => {
      if (e.key === "Enter") commit();
      else if (e.key === "Escape") cancel();
    });
    edit.addEventListener("blur", commit);
    dueEdit.addEventListener("keydown", (e) => {
      if (e.key === "Enter") commit();
      else if (e.key === "Escape") cancel();
    });

    li.appendChild(edit);
    li.appendChild(dueEdit);
    edit.focus();
    edit.select();
  }
```

Note: `commit` on `blur` plus the `done` guard prevents double-saving when focus moves between the two inputs is acceptable here because clicking the date input blurs the text input and commits; to edit the date, set it first, then edit text, or change the date and press Enter. Keep behavior simple: editing the date is done via the date input then Enter.
